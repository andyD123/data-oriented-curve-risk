cmake_minimum_required(VERSION 3.16)
project(ladder CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(LADDER_LANES "STDX" CACHE STRING "Lane backend: STDX (std::experimental::simd), AVX512, AVX2, PORTABLE, or AUTO")
add_library(ladder INTERFACE)
target_include_directories(ladder INTERFACE ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(ladder INTERFACE -march=native -mprefer-vector-width=512 -ffp-contract=fast)
if(NOT LADDER_LANES STREQUAL "STDX")
  target_compile_definitions(ladder INTERFACE LADDER_LANES_${LADDER_LANES})
endif()
add_executable(test_ladder tests/test_ladder.cpp)
target_link_libraries(test_ladder ladder)
add_test(NAME ladder COMMAND test_ladder)
enable_testing()
add_executable(bench examples/benchmark/bench.cpp)
target_link_libraries(bench ladder)
target_compile_options(bench PRIVATE -O3 -Wno-unused-result)
