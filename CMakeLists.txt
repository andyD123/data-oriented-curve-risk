  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)

set(LADDER_LANES "AUTO" CACHE STRING "Lane backend: AUTO, STDX, AVX512, AVX2, PORTABLE")
set_property(CACHE LADDER_LANES PROPERTY STRINGS AUTO STDX AVX512 AVX2 PORTABLE)
set(_lane_choices AUTO STDX AVX512 AVX2 PORTABLE)
if(NOT LADDER_LANES IN_LIST _lane_choices)
  message(FATAL_ERROR "Unknown LADDER_LANES=${LADDER_LANES}; choose ${_lane_choices}")
endif()
option(LADDER_NATIVE_ARCH "Generate code for this build host (not a portable binary)" OFF)
option(LADDER_BUILD_EXAMPLES "Build library, aggregation and wave replay examples" ON)
option(LADDER_BUILD_PAPER_BENCHMARKS "Build historical paper benchmarks (requires x86 AVX-512)" OFF)
set(LADDER_BUILD_QUANTLIB "AUTO" CACHE STRING "QuantLib generators: AUTO, ON, OFF")
set_property(CACHE LADDER_BUILD_QUANTLIB PROPERTY STRINGS AUTO ON OFF)
if(NOT LADDER_BUILD_QUANTLIB MATCHES "^(AUTO|ON|OFF)$")
  message(FATAL_ERROR "LADDER_BUILD_QUANTLIB must be AUTO, ON or OFF")
endif()

add_library(ladder INTERFACE)
add_library(ladder::ladder ALIAS ladder)
target_include_directories(ladder INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(ladder INTERFACE cxx_std_20)
target_compile_definitions(ladder INTERFACE LADDER_LANES_${LADDER_LANES})
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(ladder INTERFACE -ffp-contract=fast)
endif()
if(LADDER_NATIVE_ARCH)
  check_cxx_compiler_flag("-march=native" LADDER_HAS_MARCH_NATIVE)
  if(NOT LADDER_HAS_MARCH_NATIVE)
    message(FATAL_ERROR "Native architecture flag unavailable; use an explicit lane backend")
  endif()
  target_compile_options(ladder INTERFACE -march=native)
endif()
if(LADDER_LANES MATCHES "^AVX(2|512)$")
  if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64|x86|i.86)$")
    message(FATAL_ERROR "The selected intrinsics backend requires an x86 target")
  endif()
  if(MSVC)
    if(LADDER_LANES STREQUAL "AVX512")
      target_compile_options(ladder INTERFACE /arch:AVX512)
    else()
      target_compile_options(ladder INTERFACE /arch:AVX2)
    endif()
  elseif(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    if(LADDER_LANES STREQUAL "AVX512")
      target_compile_options(ladder INTERFACE -mavx512f -mfma)
    else()
      target_compile_options(ladder INTERFACE -mavx2 -mfma)
    endif()
  else()
    message(FATAL_ERROR "No explicit ISA flags configured for this compiler")
  endif()
endif()
if(LADDER_LANES STREQUAL "STDX")
  check_include_file_cxx("experimental/simd" LADDER_HAS_STDX)
  if(NOT LADDER_HAS_STDX)
    message(FATAL_ERROR "STDX needs <experimental/simd>; choose AUTO or PORTABLE instead")
  endif()
endif()

# Keep the shared CLion run targets and copied build-directory data from main.
function(ladder_example name dir src)
  add_executable(${name} examples/${dir}/${src})
  target_link_libraries(${name} PRIVATE ladder)
  set_target_properties(${name} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/examples/${dir}")
  target_compile_definitions(${name} PRIVATE LADDER_EXAMPLE_DIR="${CMAKE_BINARY_DIR}/examples/${dir}")
  file(GLOB _data "${CMAKE_CURRENT_SOURCE_DIR}/examples/${dir}/*.txt" "${CMAKE_CURRENT_SOURCE_DIR}/examples/${dir}/*.py")
  if(_data)
    file(COPY ${_data} DESTINATION "${CMAKE_BINARY_DIR}/examples/${dir}")
  endif()
endfunction()

# The historical standalone kernel assumes native_simd<double>::size()==8.
# Do not let that assumption break a default/portable/AVX2 library build.
if(LADDER_BUILD_PAPER_BENCHMARKS)
  if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64|x86|i.86)$"
      OR NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    message(FATAL_ERROR "Paper benchmarks require GCC/Clang on an AVX-512 x86 target")
  endif()
  check_include_file_cxx("experimental/simd" LADDER_PAPER_HAS_STDX)
  if(NOT LADDER_PAPER_HAS_STDX)
    message(FATAL_ERROR "Paper benchmark requires <experimental/simd>")
  endif()
  ladder_example(bench_paper benchmark_paper bench_paper.cpp)
  ladder_example(scenario_bench benchmark_paper scenario_bench.cpp)
  ladder_example(adjoint_bench benchmark_paper adjoint_bench.cpp)
  foreach(_target bench_paper scenario_bench adjoint_bench)
    target_compile_options(${_target} PRIVATE -mavx512f -mavx512dq -mfma -mprefer-vector-width=512)
  endforeach()
endif()

if(NOT LADDER_BUILD_QUANTLIB STREQUAL "OFF")
  find_package(QuantLib CONFIG QUIET)
  if(TARGET QuantLib::QuantLib)
    set(_quantlib QuantLib::QuantLib)
  elseif(TARGET QuantLib)
    set(_quantlib QuantLib)
  else()
    find_path(QUANTLIB_INCLUDE_DIR ql/quantlib.hpp)
    find_library(QUANTLIB_LIBRARY NAMES QuantLib quantlib)
    if(QUANTLIB_INCLUDE_DIR AND QUANTLIB_LIBRARY)
      add_library(ladder_quantlib INTERFACE)
      target_include_directories(ladder_quantlib INTERFACE "${QUANTLIB_INCLUDE_DIR}")
      target_link_libraries(ladder_quantlib INTERFACE "${QUANTLIB_LIBRARY}")
      set(_quantlib ladder_quantlib)
    endif()
  endif()
  if(_quantlib)
    ladder_example(ql_waves hagan_waves ql_waves.cpp)
    ladder_example(ql_examples quantlib_reconcile ql_examples.cpp)
    foreach(_target ql_waves ql_examples)
      target_link_libraries(${_target} PRIVATE ${_quantlib})
    endforeach()
    ladder_example(scan_reconcile quantlib_reconcile scan_reconcile.cpp)
    message(STATUS "QuantLib found: reference generators enabled")
  elseif(LADDER_BUILD_QUANTLIB STREQUAL "ON")
    message(FATAL_ERROR "QuantLib C++ headers/library not found. Install QuantLib or use AUTO/OFF.")
  else()
    message(STATUS "QuantLib not found: reference generators skipped; recorded replay remains available")
  endif()
endif()
